option(YK_SANITIZERS "Enable address/undefined sanitizers on project targets" OFF)
function(yk_target target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    endif()
    if(YK_SANITIZERS)
        if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
            message(FATAL_ERROR "YK_SANITIZERS requires Clang or GCC")
        endif()
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

add_library(yk_engine STATIC
    engine/src/core/Log.cpp engine/src/core/Time.cpp
    engine/src/input/Input.cpp engine/src/platform/Application.cpp
    engine/src/graphics/Renderer.cpp)
add_library(yk::engine ALIAS yk_engine)

target_link_libraries(yk_engine PRIVATE SDL3::SDL3)
yk_target(yk_engine)
add_executable(yk_sandbox games/sandbox/src/main.cpp)
target_link_libraries(yk_sandbox PRIVATE yk::engine)
yk_target(yk_sandbox)

        MACOSX_BUNDLE_GUI_IDENTIFIER "com.ykengine.sandbox"
        MACOSX_BUNDLE_BUNDLE_NAME "YK Sandbox")
endif()
add_custom_target(run COMMAND $<TARGET_FILE:yk_sandbox> DEPENDS yk_sandbox USES_TERMINAL)

    target_link_libraries(yk_unit_tests PRIVATE yk::engine)
